public class Main {

    public static void main(String[] args) {

        System.out.println("===== FINTECH FIAP =====");

        ContaCorrente contaCorrente = new ContaCorrente(
                1001,
                1500.00,
                "1234",
                1000.00
        );

        ContaInvestimento contaInvestimento = new ContaInvestimento(
                2001,
                5000.00,
                "1234",
                1.5
        );

        System.out.println("\n--- CONTA CORRENTE ---");
        contaCorrente.consultarSaldo();

        contaCorrente.realizarDeposito(500.00);

        contaCorrente.realizarSaque(200.00);

        contaCorrente.consultarLimite();

        System.out.println("\n--- CONTA INVESTIMENTO ---");
        contaInvestimento.consultarSaldo();

        contaInvestimento.calcularRendimento();

        contaInvestimento.realizarDeposito(1000.00);

        contaInvestimento.consultarSaldo();

        System.out.println("\n--- POLIMORFISMO ---");

        Conta conta1 = contaCorrente;
        Conta conta2 = contaInvestimento;

        conta1.consultarSaldo();

    }
}
