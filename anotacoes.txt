public class Transacao {

    private Integer idTransacao;
    private String tipoTransacao;
    private Double valor;
    private String dataTransacao;
    private boolean realizada;
    private boolean cancelada;

    public Transacao(Integer idTransacao, String tipoTransacao, Double valor, String dataTransacao) {
        this.idTransacao = idTransacao;
        this.tipoTransacao = tipoTransacao;
        this.valor = valor;
        this.dataTransacao = dataTransacao;
        this.realizada = false;
        this.cancelada = false;
    }

    public void realizarTransacao() {
        if (valor != null && valor > 0 && !cancelada) {
            realizada = true;
            System.out.println("Transação realizada com sucesso.");
        } else {
            System.out.println("Não foi possível realizar a transação.");
        }
    }

    public void consultarTransacao() {
        System.out.println("ID da transação: " + idTransacao);
        System.out.println("Tipo: " + tipoTransacao);
        System.out.println("Valor: R$ " + valor);
        System.out.println("Data: " + dataTransacao);
        System.out.println("Realizada: " + realizada);
        System.out.println("Cancelada: " + cancelada);
    }

    public void cancelarTransacao() {
        if (realizada && !cancelada) {
            cancelada = true;
            System.out.println("Transação cancelada com sucesso.");
        } else {
            System.out.println("Não foi possível cancelar a transação.");
        }
    }

    public Integer getIdTransacao() {
        return idTransacao;
    }

    public String getTipoTransacao() {
        return tipoTransacao;
    }

    public Double getValor() {
        return valor;
    }

    public String getDataTransacao() {
        return dataTransacao;
    }

    public boolean isRealizada() {
        return realizada;
    }

    public boolean isCancelada() {
        return cancelada;
    }
}
