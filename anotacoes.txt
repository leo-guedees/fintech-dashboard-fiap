public class ContaInvestimento extends Conta {

    private Double percentualRendimento;

    public ContaInvestimento(Integer numeroConta, Double saldo, String agencia, Double percentualRendimento) {
        super(numeroConta, "Conta Investimento", saldo, agencia);
        this.percentualRendimento = percentualRendimento;
    }

    @Override
    public void consultarSaldo() {
        System.out.println("Saldo da Conta Investimento: R$ " + getSaldo());
        System.out.println("Rendimento: " + percentualRendimento + "%");
    }

    public void calcularRendimento() {
        Double rendimento = getSaldo() * (percentualRendimento / 100);
        System.out.println("Rendimento estimado: R$ " + rendimento);
    }
}
