public class ContaCorrente extends Conta {

    private Double limite;

    public ContaCorrente(Integer numeroConta, Double saldo, String agencia, Double limite) {
        super(numeroConta, "Conta Corrente", saldo, agencia);
        this.limite = limite;
    }

    @Override
    public void consultarSaldo() {
        System.out.println("Saldo da Conta Corrente: R$ " + getSaldo());
        System.out.println("Limite disponível: R$ " + limite);
    }

    public void consultarLimite() {
        System.out.println("Limite disponível: R$ " + limite);
    }
}
